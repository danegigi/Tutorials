Bun ships a fast native HTTP server, WebSockets with built-in pub/sub, and raw TCP — no framework required. This lesson builds up from Bun.serve routing to real-time messaging, then adds a Hono app with JWT auth stored in HttpOnly cookies for the full web-app shape.

Bun.serve — routing

Beginner The modern API uses a routes object with path params and per-method handlers. req.params holds matched segments; a top-level fetch is the fallback.

const server = Bun.serve({
  port: 3000,
  routes: {
    "/": () => new Response("Hello, Bun!"),

    "/users/:id": (req) => Response.json({ id: req.params.id }),

    "/api/echo": {
      POST: async (req) => Response.json(await req.json()),
    },

    // multiple methods on one path
    "/api/orders/:id": {
      GET:    (req) => Response.json({ id: req.params.id, status: "ok" }),
      DELETE: (req) => new Response(null, { status: 204 }),
    },
  },
  fetch() {                                   // unmatched routes
    return new Response("Not found", { status: 404 });
  },
});

console.log(`Listening on ${server.url}`);    // http://localhost:3000

Serving files & streaming

Bun.serve({
  routes: {
    // serve straight from disk — uses sendfile under the hood
    "/logo.png": () => new Response(Bun.file("./assets/logo.png")),

    // stream a response without buffering it all in memory
    "/stream": () => new Response(
      new ReadableStream({
        start(c) { c.enqueue("chunk 1\n"); c.enqueue("chunk 2\n"); c.close(); },
      }),
      { headers: { "Content-Type": "text/plain" } },
    ),
  },
});

WebSockets + pub/sub

Intermediate Call server.upgrade(req) and return without a Response to hand the connection to the websocket handlers. Bun's built-in subscribe/publish is topic-based pub/sub — a chat room in a few lines, no Redis, no socket.io.

const server = Bun.serve({
  port: 3001,
  fetch(req, server) {
    // pass per-connection data to the socket
    const user = new URL(req.url).searchParams.get("user") ?? "anon";
    if (server.upgrade(req, { data: { user } })) return;   // upgraded
    return new Response("Use a WebSocket client");
  },
  websocket: {
    open(ws) {
      ws.subscribe("room");                       // join the topic
      server.publish("room", `${ws.data.user} joined`);
    },
    message(ws, msg) {
      server.publish("room", `${ws.data.user}: ${msg}`);   // broadcast to the topic
    },
    close(ws) {
      ws.unsubscribe("room");
      server.publish("room", `${ws.data.user} left`);
    },
  },
});

A browser client:

const ws = new WebSocket("ws://localhost:3001/?user=ada");
ws.onmessage = (e) => console.log(e.data);
ws.onopen = () => ws.send("hello room");
ws.publish(topic, msg) broadcasts to everyone except the sender; server.publish(topic, msg) includes the sender. Choose based on whether the author should see their own message echoed. Backpressure is handled for you — ws.send returns the bytes queued; don't resend on a partial write.

Raw TCP

For protocols below HTTP (a line-based service, a proxy, a game server), Bun.listen and Bun.connect give you raw sockets:

// Echo server
Bun.listen({
  hostname: "localhost",
  port: 8080,
  socket: {
    open(socket) { socket.write("hi\n"); },
    data(socket, data) { socket.write(data); },   // echo back
    close() { console.log("client gone"); },
  },
});

// Client
const client = await Bun.connect({
  hostname: "localhost", port: 8080,
  socket: { data(socket, data) { console.log("got", data.toString()); } },
});

A Hono app with JWT auth

Advanced Hono is a tiny third-party router that runs on Bun.serve (pin it: bun add hono@4.6.3). The auth pattern: a short-lived access token (~15 min) the client sends per request, and a long-lived refresh token (~7 days) in an HttpOnly; Secure; SameSite cookie JavaScript can't read — so an XSS can't steal it. /refresh rotates the pair.

import { Hono } from "hono";
import { sign, verify } from "hono/jwt";
import { setCookie, getCookie, deleteCookie } from "hono/cookie";

const app = new Hono();
const ACCESS_SECRET  = Bun.env.ACCESS_SECRET!;
const REFRESH_SECRET = Bun.env.REFRESH_SECRET!;

const now = () => Math.floor(Date.now() / 1000);
const accessToken  = (sub: string) => sign({ sub, exp: now() + 60 * 15 }, ACCESS_SECRET);
const refreshToken = (sub: string) => sign({ sub, exp: now() + 60 * 60 * 24 * 7 }, REFRESH_SECRET);

function setRefreshCookie(c: any, token: string) {
  setCookie(c, "refresh_token", token, {
    httpOnly: true,          // JS can't read it -> XSS can't exfiltrate it
    secure: true,            // HTTPS only
    sameSite: "Strict",      // not sent on cross-site requests -> CSRF mitigation
    path: "/refresh",        // only sent to the refresh endpoint
    maxAge: 60 * 60 * 24 * 7,
  });
}

// LOGIN — verify credentials (argon2id), issue both tokens
app.post("/login", async (c) => {
  const { email, password } = await c.req.json();
  const user = await findUser(email);
  if (!user || !(await Bun.password.verify(password, user.passwordHash))) {
    return c.json({ error: "invalid credentials" }, 401);
  }
  setRefreshCookie(c, await refreshToken(user.id));
  return c.json({ accessToken: await accessToken(user.id) });
});

// REFRESH — verify the cookie, rotate the pair
app.post("/refresh", async (c) => {
  const token = getCookie(c, "refresh_token");
  if (!token) return c.json({ error: "no refresh token" }, 401);
  try {
    const payload = await verify(token, REFRESH_SECRET);
    setRefreshCookie(c, await refreshToken(payload.sub as string));   // rotate
    return c.json({ accessToken: await accessToken(payload.sub as string) });
  } catch {
    return c.json({ error: "invalid refresh token" }, 401);
  }
});

// LOGOUT — clear the cookie
app.post("/logout", (c) => {
  deleteCookie(c, "refresh_token", { path: "/refresh" });
  return c.json({ ok: true });
});

// Auth middleware for protected routes — validate the access token
async function requireAuth(c: any, next: any) {
  const token = (c.req.header("Authorization") ?? "").replace(/^Bearer /, "");
  try {
    c.set("userId", (await verify(token, ACCESS_SECRET)).sub);
    await next();
  } catch {
    return c.json({ error: "unauthorized" }, 401);
  }
}

app.get("/me", requireAuth, (c) => c.json({ userId: c.get("userId") }));

export default app;      // Bun auto-serves a default export with a fetch handler
bun --hot server.ts      # http://localhost:3000

Request logging & error handling middleware

app.use("*", async (c, next) => {
  const t = Bun.nanoseconds();
  await next();
  c.header("X-Response-Time", `${((Bun.nanoseconds() - t) / 1e6).toFixed(2)}ms`);
});

app.onError((err, c) => {
  console.error(err);
  return c.json({ error: "internal" }, 500);
});

Common mistakes

Common mistake: storing the access token (or the refresh token) in localStorage. Anything in localStorage is readable by any script on the page, so an XSS steals it instantly. Keep the refresh token in an HttpOnly cookie (unreadable by JS) and the short-lived access token in memory only. Two more traps: returning a Response after server.upgrade(req) succeeds (don't — return nothing so Bun completes the upgrade), and trusting X-Forwarded-For for the client IP without a known proxy in front (it's spoofable; only trust it from your own load balancer, otherwise use server.requestIP(req)).

Exercises

  • Write a WebSocket chat where two browser tabs exchange messages via subscribe/publish on one topic.
  • Add requireAuth middleware to a Hono route and confirm a request with no Authorization header gets 401.
  • Implement /refresh rotation and verify logging out (clearing the cookie) makes the next /refresh fail.

Mini-project: a real-time orders API with auth and live updates

Combine everything: a JSON API protected by JWT, plus a WebSocket channel that pushes new orders to connected dashboards in real time, with graceful shutdown for container deploys.

// server.ts  —  run with: bun --hot server.ts
type Order = { id: string; total: number; status: string };
const orders: Order[] = [];
const SECRET = Bun.env.ACCESS_SECRET ?? "dev-secret";

const server = Bun.serve({
  port: Number(Bun.env.PORT ?? 3000),

  routes: {
    // Create an order (would require auth in prod — simplified here)
    "/api/orders": {
      POST: async (req) => {
        const body = await req.json();
        const order: Order = { id: crypto.randomUUID(), total: body.total, status: "pending" };
        orders.push(order);
        server.publish("orders", JSON.stringify({ type: "new", order }));  // live push
        return Response.json(order, { status: 201 });
      },
      GET: () => Response.json(orders),
    },
  },

  // WebSocket endpoint: a dashboard subscribes to the "orders" topic
  fetch(req, server) {
    if (new URL(req.url).pathname === "/live" && server.upgrade(req)) return;
    return new Response("not found", { status: 404 });
  },
  websocket: {
    open(ws) { ws.subscribe("orders"); ws.send(JSON.stringify({ type: "snapshot", orders })); },
    close(ws) { ws.unsubscribe("orders"); },
    message() {},   // dashboards only listen
  },

  error(err) { console.error(err); return Response.json({ error: "internal" }, { status: 500 }); },
});

// Graceful shutdown — drain in-flight requests before exiting (what k8s sends)
process.on("SIGTERM", async () => { await server.stop(); process.exit(0); });

console.log(`orders API on ${server.url}  (WS at /live)`);

Test it: POST /api/orders with a JSON body, and watch a WebSocket client connected to /live receive the new event instantly. The snapshot on connect gives a late joiner the current state; the SIGTERM handler lets a rolling deploy finish in-flight work instead of dropping it. Harden it by putting the requireAuth middleware from above in front of the write route and authenticating the WebSocket upgrade with the access token.
